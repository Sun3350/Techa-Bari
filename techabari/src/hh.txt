import React, { useState, useEffect } from "react";
import { IoIosSend } from "react-icons/io";
import ReactMarkdown from "react-markdown";
import "./App.css";

export default function Home() {
  const [userInput, setUserInput] = useState("");
  const [sessions, setSessions] = useState([]);
  const [currentSession, setCurrentSession] = useState(null);
  const [response, setResponse] = useState([]);
  const [isLoading, setIsLoading] = useState(false);

  // Fetch chat sessions for the logged-in user
  useEffect(() => {
    const fetchSessions = async () => {
      const userId = getLoggedInUserId(); // Replace with actual user ID logic
      const res = await fetch(`/api/chat/all-chats/${userId}`);
      const data = await res.json();
      setSessions(data);
    };
    fetchSessions();
  }, []);

  // Load history when a session is selected
  useEffect(() => {
    if (currentSession) {
      setResponse(currentSession.history);
    }
  }, [currentSession]);

  const handleUserInput = (e) => setUserInput(e.target.value);

  const handleCreateSession = async () => {
    const userId = getLoggedInUserId();
    const res = await fetch("/api/chat/new-chats", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ userId, sessionName: "New Chat" }),
    });
    const newSession = await res.json();
    setSessions([...sessions, newSession]);
    setCurrentSession(newSession);
  };

  const handleDeleteSession = async (sessionId) => {
    await fetch(`/api/chat/session/${sessionId}`, { method: "DELETE" });
    setSessions(sessions.filter((s) => s._id !== sessionId));
    if (currentSession?._id === sessionId) setCurrentSession(null);
  };

  const handleSubmit = async () => {
    if (!userInput.trim() || !currentSession) return;

    setIsLoading(true);
    try {
      const res = await generateContent(userInput);
      const newResponse = [
        ...response,
        { type: "user", message: userInput },
        { type: "bot", message: res() },
      ];
      setResponse(newResponse);
      setUserInput("");

      // Save updated history
      await fetch("/api/chat/save-chats/", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ sessionId: currentSession._id, history: newResponse }),
      });
    } catch (error) {
      console.error("Error generating response:", error);
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <div className="chat-app">
      {/* Sidebar */}
      <div className="chat-sidebar">
        <button onClick={handleCreateSession}>+ New Chat</button>
        {sessions.map((session) => (
          <div
            key={session._id}
            className={`chat-session ${
              currentSession?._id === session._id ? "active" : ""
            }`}
            onClick={() => setCurrentSession(session)}
          >
            <span>{session.sessionName}</span>
            <button onClick={() => handleDeleteSession(session._id)}>x</button>
          </div>
        ))}
      </div>

      {/* Chat Window */}
      <div className="chat-window">
        {currentSession ? (
          <>
            <div className="chat-history">
              {response.map((msg, index) => (
                <p key={index} className={`message ${msg.type}`}>
                  <ReactMarkdown>{msg.message}</ReactMarkdown>
                </p>
              ))}
              {isLoading && <p>Loading...</p>}
            </div>
            <div className="chat-input-container">
              <input
                value={userInput}
                onChange={handleUserInput}
                placeholder="Type your message here..."
              />
              <button onClick={handleSubmit}>
                <IoIosSend />
              </button>
            </div>
          </>
        ) : (
          <p>Select or create a chat session to begin.</p>
        )}
      </div>
    </div>
  );
}
